"""Educational multimodal contrastive and cross-attention primitives."""

from __future__ import annotations

import numpy as np


def l2_normalize(
    x: np.ndarray,
    axis: int = -1,
    eps: float = 1e-12,
) -> np.ndarray:
    values = np.asarray(x, dtype=float)
    norm = np.linalg.norm(values, axis=axis, keepdims=True)
    return values / np.maximum(norm, eps)


def similarity_logits(
    left: np.ndarray,
    right: np.ndarray,
    *,
    temperature: float = 0.07,
) -> np.ndarray:
    if temperature <= 0:
        raise ValueError("temperature must be positive")

    a = np.asarray(left, dtype=float)
    b = np.asarray(right, dtype=float)

    if a.ndim != 2 or b.ndim != 2:
        raise ValueError("embeddings must be matrices")
    if a.shape[1] != b.shape[1]:
        raise ValueError("embedding dimensions must match")

    a = l2_normalize(a)
    b = l2_normalize(b)
    return (a @ b.T) / temperature


def stable_cross_entropy(
    logits: np.ndarray,
    targets: np.ndarray,
) -> float:
    scores = np.asarray(logits, dtype=float)
    y = np.asarray(targets, dtype=np.int64)

    if scores.ndim != 2 or y.shape != (scores.shape[0],):
        raise ValueError("shape mismatch")
    if np.any((y < 0) | (y >= scores.shape[1])):
        raise ValueError("target out of range")

    maximum = np.max(scores, axis=1, keepdims=True)
    logsumexp = (
        maximum[:, 0]
        + np.log(np.sum(np.exp(scores - maximum), axis=1))
    )
    correct = scores[np.arange(len(y)), y]
    return float(np.mean(logsumexp - correct))


def symmetric_contrastive_loss(
    image_embeddings: np.ndarray,
    text_embeddings: np.ndarray,
    *,
    temperature: float = 0.07,
) -> float:
    images = np.asarray(image_embeddings, dtype=float)
    texts = np.asarray(text_embeddings, dtype=float)

    if images.shape != texts.shape:
        raise ValueError("paired embedding matrices must have equal shape")

    logits = similarity_logits(
        images,
        texts,
        temperature=temperature,
    )
    targets = np.arange(len(images), dtype=np.int64)

    return 0.5 * (
        stable_cross_entropy(logits, targets)
        + stable_cross_entropy(logits.T, targets)
    )


def retrieval_top1_accuracy(
    query_embeddings: np.ndarray,
    candidate_embeddings: np.ndarray,
) -> float:
    q = np.asarray(query_embeddings, dtype=float)
    c = np.asarray(candidate_embeddings, dtype=float)

    if q.shape != c.shape:
        raise ValueError("paired matrices must have equal shape")

    logits = similarity_logits(q, c, temperature=1.0)
    predicted = np.argmax(logits, axis=1)
    expected = np.arange(len(q))
    return float(np.mean(predicted == expected))


def simple_cross_attention(
    query: np.ndarray,
    key: np.ndarray,
    value: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    q = np.asarray(query, dtype=float)
    k = np.asarray(key, dtype=float)
    v = np.asarray(value, dtype=float)

    if q.ndim != 3 or k.ndim != 3 or v.ndim != 3:
        raise ValueError("expected batched Q/K/V tensors")
    if q.shape[0] != k.shape[0] or k.shape[0] != v.shape[0]:
        raise ValueError("batch sizes must match")
    if q.shape[-1] != k.shape[-1]:
        raise ValueError("query/key dimensions must match")
    if k.shape[1] != v.shape[1]:
        raise ValueError("key/value sequence lengths must match")

    scores = q @ np.swapaxes(k, -1, -2)
    scores = scores / np.sqrt(q.shape[-1])
    scores = scores - np.max(scores, axis=-1, keepdims=True)

    weights = np.exp(scores)
    weights = weights / np.sum(weights, axis=-1, keepdims=True)
    return weights @ v, weights

ส่วนของ feature representation ที่ทำ attention แยกกัน แล้ว concat/project กลับ; ต้องตรวจ d_model หารจำนวน heads ลงตัว
ε-neighborhood
เพื่อนบ้านทั้งหมดที่มีระยะไม่เกิน ε รวมจุดตัวเองใน DBSCAN นี้; ε มีหน่วยเดียวกับพิกัด ไม่ใช่จำนวน k ที่ใช้โหวต
core / border / noise
DBSCAN core มีเพื่อนบ้านรวมตัวเองถึง min_samples; border อยู่ใกล้ core แต่เพื่อนบ้านไม่ถึง; noise ไม่ถูกเชื่อมกับ core cluster ใด
bootstrap
สุ่ม observations แบบ replacement ขนาดเท่าเดิมซ้ำหลายครั้งแล้วคำนวณ statistic ใหม่; seed ทำให้ตรวจซ้ำได้ แต่ไม่แก้สมมติฐาน IID ที่ผิด
confidence interval
ช่วงที่สร้างด้วย procedure ที่มี coverage ตามสมมติฐาน; ช่วงหนึ่งที่ได้แล้วไม่ใช่ posterior probability ของ parameter และ sample เล็กยังไม่แน่นอนมาก
validation / test
validation ใช้เลือกโมเดล/ค่าตั้ง ส่วน test ใช้ประเมินหลัง freeze การตัดสินใจ; fit preprocessing/tokenizer จาก test เป็น leakage
leakage
ข้อมูลที่ควรยังไม่รู้ไหลเข้า training/การเลือกโมเดล เช่น future feature หรือเอกสารซ้ำข้าม split ทำให้ metric ดูดีเกินการใช้งานจริง
queue / capacity
queue คือรายการรอ, capacity การให้บริการรวม = workers × service rate ตามแบบจำลอง; เพิ่ม timeout ไม่เพิ่มกำลังประมวลผล
state transition
การเปลี่ยนสถานะเมื่อเกิดเหตุการณ์ที่อนุญาต เช่น queued→active→done; ต้องระบุ guard, side effect และเหตุผลเมื่อ transition ถูกปฏิเสธ
latency / throughput
latency คือเวลาต่อคำขอ, throughput คือปริมาณงานต่อหน่วยเวลา; batching อาจเพิ่ม throughput แต่เพิ่มเวลารอ จึงต้องวัดทั้งคู่
idempotency
ทำคำขอ business operation เดิมซ้ำแล้ว side effect เกิดครั้งเดียวตาม key และ transaction protocol; ไม่ได้รับประกันจากการใช้ UUID ใหม่ทุก attempt
local lab
โปรแกรม Python และ tests ที่ผู้เรียนรันบนเครื่อง; เว็บเก็บเพียงคำยืนยันของผู้เรียน การผ่าน tests ของโค้ดต้นแบบไม่ใช่ผลตรวจโค้ดผู้เรียน

Extension: - add token-level cross-attention and a small classification task requiring both modalities.

Batch 14 Integration Project — Diffusion, Multimodal & Speech Lab

Batch 14 contains three framework experiments.

Part A — 2D DDPM

4-mode 2D data
↓
sample random timestep
↓
add Gaussian noise
↓
MLP epsilon predictor
↓
MSE noise objective
↓
reverse DDPM sampler

Tracks: - training loss - parameter count - generated mean/covariance - number of diffusion steps

Part B — CLIP-Style Alignment

Two synthetic modalities are generated from the same hidden semantic vector.

image features → image encoder ─┐
                               ├→ contrastive loss
text features  → text encoder ─┘

Tracks: - symmetric contrastive loss - image→text top-1 retrieval - text→image top-1 retrieval

Part C — Synthetic Speech CTC

toy token sequence
↓
tone waveform
↓
torch.stft + log power
↓
BiGRU acoustic model
↓
CTC logits
↓
greedy CTC decode

This verifies: - waveform frontend - explicit Hann window - complex STFT output - CTC log-probability shapes - blank-token decoding

It is not a real human speech recognizer.

Run

python integration-project-batch14/src/diffusion_multimodal_speech_lab.py --mode diffusion --steps 100
python integration-project-batch14/src/diffusion_multimodal_speech_lab.py --mode multimodal --steps 100
python integration-project-batch14/src/diffusion_multimodal_speech_lab.py --mode speech --steps 40

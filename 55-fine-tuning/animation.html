3. Pretraining vs Fine-Tuning

"""Framework-independent fine-tuning planning utilities."""

from __future__ import annotations

from dataclasses import dataclass


def parameter_partition(
    named_parameter_sizes: dict[str, int],
    trainable_names: set[str],
) -> tuple[int, int]:
    if any(size < 0 for size in named_parameter_sizes.values()):
        raise ValueError("parameter sizes must be non-negative")

    total = sum(named_parameter_sizes.values())
    trainable = sum(
        size
        for name, size in named_parameter_sizes.items()
        if name in trainable_names
    )
    return trainable, total


def trainable_ratio(
    trainable_parameters: int,
    total_parameters: int,
) -> float:
    if trainable_parameters < 0:
        raise ValueError("trainable_parameters must be non-negative")
    if total_parameters <= 0:
        raise ValueError("total_parameters must be positive")
    if trainable_parameters > total_parameters:
        raise ValueError("trainable cannot exceed total")

    return float(trainable_parameters / total_parameters)


def freeze_by_prefix(
    parameter_names: list[str],
    prefixes: tuple[str, ...],
) -> set[str]:
    """Return names that remain trainable after freezing prefixes."""
    return {
        name
        for name in parameter_names
        if not name.startswith(prefixes)
    }


def unfreeze_by_prefix(
    parameter_names: list[str],
    prefixes: tuple[str, ...],
) -> set[str]:
    """Return only names whose prefix should be trainable."""
    return {
        name
        for name in parameter_names
        if name.startswith(prefixes)
    }


def discriminative_lr_groups(
    parameter_names: list[str],
    *,
    rules: list[tuple[str, float]],
    default_lr: float,
) -> dict[str, float]:
    if default_lr <= 0:
        raise ValueError("default_lr must be positive")
    if any(lr <= 0 for _, lr in rules):
        raise ValueError("all learning rates must be positive")

    result: dict[str, float] = {}

    for name in parameter_names:
        learning_rate = default_lr

        for prefix, candidate_lr in rules:
            if name.startswith(prefix):
                learning_rate = candidate_lr
                break

        result[name] = float(learning_rate)

    return result


@dataclass(frozen=True)
class EarlyStoppingState:
    best_metric: float | None = None
    bad_epochs: int = 0
    should_stop: bool = False


def early_stopping_update(
    state: EarlyStoppingState,
    metric: float,
    *,
    patience: int,
    mode: str = "min",
    min_delta: float = 0.0,
) -> EarlyStoppingState:
    if patience < 0:
        raise ValueError("patience must be non-negative")
    if mode not in {"min", "max"}:
        raise ValueError("mode must be min or max")
    if min_delta < 0:
        raise ValueError("min_delta must be non-negative")

    if state.best_metric is None:
        improved = True
    elif mode == "min":
        improved = metric < state.best_metric - min_delta
    else:
        improved = metric > state.best_metric + min_delta

    if improved:
        return EarlyStoppingState(
            best_metric=float(metric),
            bad_epochs=0,
            should_stop=False,
        )

    bad_epochs = state.bad_epochs + 1
    return EarlyStoppingState(
        best_metric=state.best_metric,
        bad_epochs=bad_epochs,
        should_stop=bad_epochs > patience,
    )


def select_best_checkpoint(
    metrics: list[float],
    *,
    mode: str = "min",
) -> int:
    if not metrics:
        raise ValueError("metrics must be non-empty")
    if mode not in {"min", "max"}:
        raise ValueError("mode must be min or max")

    if mode == "min":
        return min(range(len(metrics)), key=metrics.__getitem__)
    return max(range(len(metrics)), key=metrics.__getitem__)

ประมาณ derivative ด้วย [f(x+h)−f(x−h)]/(2h); h>0, ใช้ smooth point และลองหลาย h เพื่อแยก truncation error จาก roundoff
softmax
แปลง logits เป็นความน่าจะเป็นตามแกนที่กำหนด; ลบ max ก่อน exp เพื่อความเสถียร และ fully masked row ต้องมีนโยบายชัดเจน
logit
คะแนนก่อน sigmoid/softmax ไม่ใช่ probability; อาจเป็นจำนวนจริงติดลบหรือเกิน1ได้ จึงห้ามส่งให้ loss ที่คาด probability โดยไม่แปลง
temperature τ / T
ตัวหาร logits ก่อน softmax; ต้องเป็นบวก ค่าเล็กทำ distribution แหลมขึ้น โดยไม่เท่ากับ retrain weights
mask
ตัวกำหนดตำแหน่งที่อ่านหรือคิด loss ได้; causal mask ปิดอนาคต, padding mask ปิดข้อมูลเติม, loss mask เลือก target แต่ละอย่างคนละหน้าที่
Q / K / V
query ใช้ค้นข้อมูล, key ใช้จับคู่ query, value เป็นข้อมูลที่ถูกถัวเฉลี่ย; เป็น learned projections ของ representation ไม่ใช่ป้ายความหมายตายตัว
head
ส่วนของ feature representation ที่ทำ attention แยกกัน แล้ว concat/project กลับ; ต้องตรวจ d_model หารจำนวน heads ลงตัว
ε-neighborhood
เพื่อนบ้านทั้งหมดที่มีระยะไม่เกิน ε รวมจุดตัวเองใน DBSCAN นี้; ε มีหน่วยเดียวกับพิกัด ไม่ใช่จำนวน k ที่ใช้โหวต
core / border / noise
DBSCAN core มีเพื่อนบ้านรวมตัวเองถึง min_samples; border อยู่ใกล้ core แต่เพื่อนบ้านไม่ถึง; noise ไม่ถูกเชื่อมกับ core cluster ใด
bootstrap
สุ่ม observations แบบ replacement ขนาดเท่าเดิมซ้ำหลายครั้งแล้วคำนวณ statistic ใหม่; seed ทำให้ตรวจซ้ำได้ แต่ไม่แก้สมมติฐาน IID ที่ผิด
confidence interval
ช่วงที่สร้างด้วย procedure ที่มี coverage ตามสมมติฐาน; ช่วงหนึ่งที่ได้แล้วไม่ใช่ posterior probability ของ parameter และ sample เล็กยังไม่แน่นอนมาก
validation / test
validation ใช้เลือกโมเดล/ค่าตั้ง ส่วน test ใช้ประเมินหลัง freeze การตัดสินใจ; fit preprocessing/tokenizer จาก test เป็น leakage
leakage
ข้อมูลที่ควรยังไม่รู้ไหลเข้า training/การเลือกโมเดล เช่น future feature หรือเอกสารซ้ำข้าม split ทำให้ metric ดูดีเกินการใช้งานจริง
queue / capacity
queue คือรายการรอ, capacity การให้บริการรวม = workers × service rate ตามแบบจำลอง; เพิ่ม timeout ไม่เพิ่มกำลังประมวลผล
state transition
การเปลี่ยนสถานะเมื่อเกิดเหตุการณ์ที่อนุญาต เช่น queued→active→done; ต้องระบุ guard, side effect และเหตุผลเมื่อ transition ถูกปฏิเสธ
latency / throughput
latency คือเวลาต่อคำขอ, throughput คือปริมาณงานต่อหน่วยเวลา; batching อาจเพิ่ม throughput แต่เพิ่มเวลารอ จึงต้องวัดทั้งคู่
idempotency
ทำคำขอ business operation เดิมซ้ำแล้ว side effect เกิดครั้งเดียวตาม key และ transaction protocol; ไม่ได้รับประกันจากการใช้ UUID ใหม่ทุก attempt
local lab
โปรแกรม Python และ tests ที่ผู้เรียนรันบนเครื่อง; เว็บเก็บเพียงคำยืนยันของผู้เรียน การผ่าน tests ของโค้ดต้นแบบไม่ใช่ผลตรวจโค้ดผู้เรียน

A frozen synthetic pretrained linear map is copied into a trainable model.

The target task differs by a known low-rank weight update.

base W
↓ full parameter update
target W'

Track: - trainable parameters - initial/final adaptation loss - retained base-task loss

W frozen
+
(alpha/r) B A
↓
adaptation

The lab verifies: - base W has requires_grad=False - B starts at zero - only A/B are optimized - trainable fraction is much smaller than full tuning

Part C — Instruction SFT

Synthetic chat-like token sequences contain:

user prompt tokens
↓ ignored labels

assistant response tokens
↓ supervised labels

A tiny causal GRU language model performs next-token SFT with ignore_index=-100.

This part tests the data/loss mechanics of instruction tuning independently from model architecture.
